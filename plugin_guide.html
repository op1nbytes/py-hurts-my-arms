1) Plugin Structure

Each plugin is a folder inside plugins/ with a manifest.json and main.py (or plugin.py).

plugins/
  my_plugin/
    manifest.json
    main.py

2) manifest.json

Minimal:

{
  "name": "MyPlugin",
  "version": "1.0",
  "settings": {}
}

With settings:

{
  "name": "MyPlugin",
  "version": "1.0",
  "settings": {
    "greeting": "Hello",
    "count": 3
  }
}

Settings show up automatically in Plugins ? Plugin Manager ? Settings.

3) Commands

Commands appear in the Command Palette as PluginName: Command.

from PyQt5.QtWidgets import QMessageBox

class Plugin:
    def __init__(self, ctx):
        self.ctx = ctx
        self.plugin_name = "MyPlugin"
        self.ctx.register_plugin_command(self.plugin_name, "Say Hello", self.say_hello)

    def say_hello(self):
        QMessageBox.information(self.ctx, "Hello", "Hello from plugin!")

4) Command Palette

Open with Ctrl+Shift+P. Every plugin command shows up here automatically.

5) Keybinds

Add keybinds in keybinds.json using:

{
  "plugin::MyPlugin: Say Hello": "Ctrl+Alt+H"
}

6) Plugin Settings API

msg = self.ctx.get_plugin_setting(self.plugin_name, "greeting", "Hi")
self.ctx.set_plugin_setting(self.plugin_name, "greeting", "Yo")

Settings are saved in plugins_settings.json.

7) UI: Add a Bottom Tab

from PyQt5.QtWidgets import QWidget, QVBoxLayout, QLabel

class Plugin:
    def __init__(self, ctx):
        self.ctx = ctx

    def setup(self):
        tab = QWidget()
        layout = QVBoxLayout()
        layout.addWidget(QLabel("Hello from a plugin tab"))
        tab.setLayout(layout)
        self.ctx.add_tab(tab, "My Tab")

8) UI: Add a Sidebar Button

from PyQt5.QtWidgets import QPushButton

class Plugin:
    def __init__(self, ctx):
        self.ctx = ctx

    def setup(self):
        btn = QPushButton("N")
        btn.setObjectName("sidebar_plugin_button")
        btn.setFixedSize(30, 30)
        btn.clicked.connect(self.open_tab)
        self.ctx.add_sidebar_widget(btn)

    def open_tab(self):
        for i in range(self.ctx.bottom_tabs.count()):
            if self.ctx.bottom_tabs.tabText(i) == "My Tab":
                self.ctx.bottom_tabs.setCurrentIndex(i)
                return

9) UI: Add a Menu

class Plugin:
    def __init__(self, ctx):
        self.ctx = ctx

    def setup(self):
        menu = self.ctx.add_menu("MyPlugin")
        action = menu.addAction("Do Thing")
        action.triggered.connect(self.do_thing)

    def do_thing(self):
        self.ctx.statusBar().showMessage("Menu clicked", 2000)

plugins/example_plugin/main.py
from PyQt5.QtWidgets import QWidget, QVBoxLayout, QLabel, QPushButton, QMessageBox

class Plugin:
    def __init__(self, ctx):
        self.ctx = ctx
        self.plugin_name = "ExamplePlugin"
        self.ctx.register_plugin_command(self.plugin_name, "Show Message", self.show_msg)

    def setup(self):
        tab = QWidget()
        layout = QVBoxLayout()
        label = QLabel("Example Plugin Tab")
        btn = QPushButton("Open Message")
        btn.clicked.connect(self.show_msg)
        layout.addWidget(label)
        layout.addWidget(btn)
        tab.setLayout(layout)
        self.ctx.add_tab(tab, "Example")

    def show_msg(self):
        msg = self.ctx.get_plugin_setting(self.plugin_name, "message", "Hi")
        QMessageBox.information(self.ctx, "Message", msg)

12) Troubleshooting

Plugin not loading: Check Plugins ? Plugin Errors.

Command missing: Ensure register_plugin_command is called in __init__.

Settings missing: Confirm settings is in manifest.json.
